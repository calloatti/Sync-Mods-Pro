using System;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using Timberborn.CoreUI;
using Timberborn.Localization;
using Timberborn.Modding;
using Timberborn.SaveMetadataSystem;
using UnityEngine;

namespace Calloatti.SyncModsPro
{
  public static class ModSyncEngine
  {
    private static readonly HashSet<string> ObsoleteModIds = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
    {
      "calloatti.syncmods",
      "Calloatti.LoadGameModValidator"
    };

    public static void ExecuteSyncMacro(ModRepository modRepository, SaveMetadata metadata, bool isStrict)
    {
      if (metadata == null || metadata.Mods == null) return;

      try
      {
        List<Mod> allMods = modRepository.Mods.ToList();
        HashSet<string> saveModIds = new HashSet<string>(metadata.Mods.Select(m => m.Id));
        HashSet<string> gameEnabledModIds = new HashSet<string>(modRepository.EnabledMods.Select(m => m.Manifest.Id));
        HashSet<string> processedIds = new HashSet<string>();

        // Strip obsolete mods from active consideration
        gameEnabledModIds.RemoveWhere(id => ObsoleteModIds.Contains(id));

        // --- GROUP 1: Saved Game Mods (Start: 2,980,000 | Step: -10,000) ---
        int savePriority = 2980000;
        foreach (var savedMod in metadata.Mods)
        {
          if (ObsoleteModIds.Contains(savedMod.Id)) continue;

          var targetMod = allMods
            .Where(m => string.Equals(m.Manifest.Id, savedMod.Id, StringComparison.OrdinalIgnoreCase))
            .OrderByDescending(m => m.ModDirectory.IsUserMod)
            .FirstOrDefault();

          if (targetMod != null)
          {
            ModPlayerPrefsHelper.ToggleMod(true, targetMod);
            ModPlayerPrefsHelper.SetModPriority(targetMod, savePriority);
            processedIds.Add(targetMod.Manifest.Id);
            savePriority -= 10000;
          }
        }

        // --- GROUP 2: Extra Enabled Mods (Start: 2,000,000 | Step: -10,000) ---
        var extraEnabledMods = allMods
          .Where(m => gameEnabledModIds.Contains(m.Manifest.Id) && !saveModIds.Contains(m.Manifest.Id))
          .GroupBy(m => m.Manifest.Id)
          .Select(g => g.First())
          .ToList();

        if (isStrict)
        {
          foreach (var mod in extraEnabledMods)
          {
            ModPlayerPrefsHelper.ToggleMod(false, mod);
          }
        }
        else
        {
          var orderedExtra = extraEnabledMods
            .OrderByDescending(m => ModPlayerPrefsHelper.GetModPriority(m))
            .ToList();

          int extraPriority = 2000000;
          foreach (var mod in orderedExtra)
          {
            ModPlayerPrefsHelper.ToggleMod(true, mod);
            ModPlayerPrefsHelper.SetModPriority(mod, extraPriority);
            processedIds.Add(mod.Manifest.Id);
            extraPriority -= 10000;
          }
        }

        // --- GROUP 3: Remaining Disabled Mods (Start: 1,000,000 | Step: -10,000) ---
        var disabledMods = allMods
          .Where(m => !processedIds.Contains(m.Manifest.Id))
          .GroupBy(m => m.Manifest.Id)
          .Select(g => g.OrderByDescending(m => m.ModDirectory.IsUserMod).First())
          .OrderBy(m => m.DisplayName)
          .ToList();

        int disabledPriority = 1000000;
        foreach (var mod in disabledMods)
        {
          ModPlayerPrefsHelper.ToggleMod(false, mod);
          ModPlayerPrefsHelper.SetModPriority(mod, disabledPriority);
          disabledPriority -= 10000;
        }

        // --- GROUP 4: Core System Overrides ---
        EnsureEssentialSystemMod(allMods, "Harmony", 3000000);
        EnsureEssentialSystemMod(allMods, "calloatti.syncmodspro", 2990000);

        PlayerPrefs.Save();
      }
      catch (Exception ex)
      {
        Debug.LogError($"[SyncModsPro] CRITICAL ERROR during execution: {ex}");
      }

      // --- POST-SYNC POPUP EXECUTION ---
      if (UnifiedModListDialog.Instance != null)
      {
        FieldInfo dialogShowerField = typeof(UnifiedModListDialog).GetField("_dialogBoxShower", BindingFlags.NonPublic | BindingFlags.Instance);
        FieldInfo locField = typeof(UnifiedModListDialog).GetField("_loc", BindingFlags.NonPublic | BindingFlags.Instance);

        DialogBoxShower dialogBoxShower = (DialogBoxShower)dialogShowerField?.GetValue(UnifiedModListDialog.Instance);
        ILoc loc = (ILoc)locField?.GetValue(UnifiedModListDialog.Instance);

        if (dialogBoxShower != null && loc != null)
        {
          var builder = dialogBoxShower.Create()
            .SetMessage(loc.T("Calloatti.SyncModsPro.SyncDialogboxText"))
            .SetConfirmButton(() => { }, loc.T("Calloatti.SyncModsPro.Button.OK"));

          builder.Show();
        }
      }
    }

    private static void EnsureEssentialSystemMod(List<Mod> allMods, string modId, int absolutePriority)
    {
      var systemMod = allMods.FirstOrDefault(m => string.Equals(m.Manifest.Id, modId, StringComparison.OrdinalIgnoreCase));
      if (systemMod != null)
      {
        ModPlayerPrefsHelper.ToggleMod(true, systemMod);
        ModPlayerPrefsHelper.SetModPriority(systemMod, absolutePriority);
      }
    }
  }
}
